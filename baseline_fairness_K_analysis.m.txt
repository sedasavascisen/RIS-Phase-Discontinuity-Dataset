%  baseline_fairness_K_analysis_VERIFIED.m

clear; close all; clc;
 
%% ---- Load dataset ----
datafile = 'RIS_dataset_large_2D.mat';
if ~exist(datafile, 'file')
    error('%s not found. Run RIS_dataset_large_2D.m first.', datafile);
end
S = load(datafile);
F = S.F; G = S.G; h_d = S.h_d;
 
fprintf('Dataset loaded: %d samples, N=%d\n', size(F,1), size(F,2));
fprintf('NOTE: Running CD at K=50 on the full 40,000-sample dataset can\n');
fprintf('take a long time (CD has O(K*N^2) complexity). A subsample is\n');
fprintf('used below for a quick check - for a full run, set subsampleSize\n');
fprintf('to size(F,1).\n\n');
 
subsampleSize = 2000;   % quick verification; set to size(F,1) for a full run
if subsampleSize < size(F,1)
    rng(0);
    idx = randperm(size(F,1), subsampleSize);
    F_sub = F(idx,:); G_sub = G(idx,:); h_d_sub = h_d(idx);
else
    F_sub = F; G_sub = G; h_d_sub = h_d;
end
 
%% ---- Actually run ris_CD.m at K=5,20,50 ----
K_vals = [5, 20, 50];
RMSE_CD_measured = zeros(1, numel(K_vals));
TIME_CD_measured = zeros(1, numel(K_vals));
 
for i = 1:numel(K_vals)
    fprintf('--- Running CD: K=%d (n=%d samples) ---\n', K_vals(i), subsampleSize);
    [~, ~, rmse_deg, t_ms] = ris_CD(F_sub, G_sub, h_d_sub, K_vals(i));
    RMSE_CD_measured(i) = rmse_deg;
    TIME_CD_measured(i) = t_ms;
end
 
%% ---- Compare against the paper's published values ----
RMSE_CD_paper = [3.50, 3.49, 3.20];
TIME_CD_paper = [0.15, 0.60, 1.50];
 
fprintf('\n=====================================================================\n');
fprintf(' VERIFICATION: Measured (this run) vs Published in Paper (Figure 2)\n');
fprintf('=====================================================================\n');
fprintf('%-6s %14s %14s %10s   %14s %14s %10s\n', ...
    'K', 'RMSE measured', 'RMSE paper', 'diff[deg]', ...
    'Time measured', 'Time paper', 'diff[ms]');
fprintf('---------------------------------------------------------------------\n');
for i = 1:numel(K_vals)
    fprintf('K=%-4d %14.2f %14.2f %10.2f   %14.3f %14.3f %10.3f\n', ...
        K_vals(i), RMSE_CD_measured(i), RMSE_CD_paper(i), ...
        abs(RMSE_CD_measured(i)-RMSE_CD_paper(i)), ...
        TIME_CD_measured(i), TIME_CD_paper(i), ...
        abs(TIME_CD_measured(i)-TIME_CD_paper(i)));
end
fprintf('=====================================================================\n');
 
max_rmse_diff = max(abs(RMSE_CD_measured - RMSE_CD_paper));
if max_rmse_diff < 0.15
    fprintf('\nRESULT: Measured RMSE values are CONSISTENT with the paper (<0.15 deg diff).\n');
    fprintf('The numbers in Figure 2 appear reliable.\n');
else
    fprintf('\nWARNING: Measured RMSE values deviate from the paper by up to %.2f deg.\n', max_rmse_diff);
    fprintf('This could be due to the subsample size (n=%d, not the full dataset)\n', subsampleSize);
    fprintf('or a different random seed - rerun with the full dataset\n');
    fprintf('(subsampleSize = size(F,1)) to confirm.\n');
end
fprintf('\nTIMING NOTE: Latency measurements are highly hardware-dependent (the\n');
fprintf('paper used an Intel Core i7-9700K system, which may differ from\n');
fprintf('your machine); large timing differences are less concerning than\n');
fprintf('RMSE differences.\n');
 
%% ---- Visualization (verified data) ----
fig = figure('Color', 'white', 'Position', [100, 100, 900, 540]);
 
subplot(1,2,1);
plot(K_vals, RMSE_CD_paper, '-ob', 'LineWidth', 2, 'MarkerFaceColor', 'b', ...
    'DisplayName', 'Published in paper'); hold on;
plot(K_vals, RMSE_CD_measured, '--sr', 'LineWidth', 2, 'MarkerFaceColor', 'r', ...
    'DisplayName', 'Measured in this run');
xlabel('K (iteration count)'); ylabel('RMSE [deg]');
title('CD RMSE: Published vs Verified');
legend('Location', 'best'); grid on;
 
subplot(1,2,2);
plot(K_vals, TIME_CD_paper, '-ob', 'LineWidth', 2, 'MarkerFaceColor', 'b', ...
    'DisplayName', 'Published in paper'); hold on;
plot(K_vals, TIME_CD_measured, '--sr', 'LineWidth', 2, 'MarkerFaceColor', 'r', ...
    'DisplayName', 'Measured in this run');
xlabel('K (iteration count)'); ylabel('Latency [ms]');
title('CD Latency: Published vs Verified');
legend('Location', 'best'); grid on;
 
sgtitle('baseline\_fairness\_K\_analysis - Independent Verification', 'FontSize', 13);
saveas(fig, 'baseline_fairness_verification.png');
 
save('baseline_fairness_verified_results.mat', ...
    'K_vals', 'RMSE_CD_measured', 'TIME_CD_measured', ...
    'RMSE_CD_paper', 'TIME_CD_paper', 'subsampleSize');
 
fprintf('\nSaved: baseline_fairness_verification.png, baseline_fairness_verified_results.mat\n');
